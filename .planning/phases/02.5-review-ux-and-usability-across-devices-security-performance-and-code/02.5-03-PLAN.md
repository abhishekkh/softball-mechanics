export const IDEAL_RANGES = {
  elbowSlot:    { min: 70, max: 110 },    // Elbow Drop if < 70, Elbow Too High if > 110
  shoulderTilt: { min: -20, max: 20 },    // Excessive Shoulder Tilt if |tilt| > 20
  hipRotation:  { min: 75, max: 105 },    // Early Hip Rotation if < 75
} as const

export function flagMechanics(
  elbowSlot: number | null,
  shoulderTilt: number | null,
  hipRotation: number | null,
  landmarks: NormalizedLandmark[],
  motionType: MotionType = 'hitting'
): MechanicsFlag[]

// Hitting flags:
// - 'Elbow Drop' (severity: 'warning', jointIndices: [RIGHT_ELBOW=14]) — elbowSlot < 70 AND armConf >= 0.70
// - 'Elbow Too High' (severity: 'warning', jointIndices: [RIGHT_ELBOW=14]) — elbowSlot > 110 AND armConf >= 0.70
// - 'Excessive Shoulder Tilt' (severity: 'warning', jointIndices: [11,12]) — |shoulderTilt| > 20 AND armConf >= 0.70
// - 'Early Hip Rotation' (severity: 'error', jointIndices: [23,24]) — hipRotation < 75 AND hipConf >= 0.70
// - 'Bat Casting' (severity: 'warning', jointIndices: [15,13]) — wristDistFromHip/shoulderWidth > 0.6 AND leadWrist.visibility >= 0.70 AND leadHip.visibility >= 0.70

// armConf is average of landmarks[12].visibility, landmarks[14].visibility, landmarks[16].visibility
// hipConf is average of landmarks[23].visibility, landmarks[24].visibility

// Bat Casting uses:
//   leadWrist = landmarks[15] (LEFT_WRIST — lead side for RHH)
//   leadHip = landmarks[23] (LEFT_HIP)
//   leftShoulder = landmarks[11], rightShoulder = landmarks[12]
//   shoulderWidth = |rightShoulder.x - leftShoulder.x| — must be > 0.05
//   wristDistFromHip = |leadWrist.x - leadHip.x|
//   flags if wristDistFromHip / shoulderWidth > 0.6
```

export function angleBetweenThreePoints(A: Point, B: Point, C: Point): number
// Returns angle at B in [0, 180] degrees
// A-B-C where B is the joint
// Collinear (180°): A=(0,0), B=(0.5,0), C=(1,0) → 180°
// Right angle (90°): A=(0,0), B=(0,1), C=(1,1) → 90°
// Degenerate (magBA=0): returns 0

export function computeElbowSlot(landmarks: NormalizedLandmark[], handedness?: 'right' | 'left'): number | null
// Returns null if any of shoulder/elbow/wrist visibility < VISIBILITY_THRESHOLD (0.65)
// Default handedness: 'right' — uses RIGHT_SHOULDER(12), RIGHT_ELBOW(14), RIGHT_WRIST(16)

export function computeShoulderTilt(landmarks: NormalizedLandmark[]): number | null
// Returns null if LEFT_SHOULDER(11) or RIGHT_SHOULDER(12) visibility < 0.65
// Returns atan2(R.y - L.y, R.x - L.x) in degrees — positive = right shoulder lower

export function computeHipRotation(landmarks: NormalizedLandmark[]): number | null
// Returns null if LEFT_HIP(23) or RIGHT_HIP(24) visibility < 0.65
// Same formula as shoulder tilt but for hips
```
